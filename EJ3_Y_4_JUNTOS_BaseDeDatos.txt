CREATE SCHEMA Zapateria;

SET SEARCH_PATH = Zapateria;

DROP TABLE IF EXISTS auditoria CASCADE;
DROP TABLE IF EXISTS posee CASCADE;
DROP TABLE IF EXISTS contiene CASCADE;
DROP TABLE IF EXISTS tiene CASCADE;
DROP TABLE IF EXISTS talle CASCADE;
DROP TABLE IF EXISTS zapato CASCADE;
DROP TABLE IF EXISTS mcelular CASCADE;
DROP TABLE IF EXISTS proveedor CASCADE;
DROP TABLE IF EXISTS venta CASCADE;
DROP TABLE IF EXISTS horario CASCADE;
DROP TABLE IF EXISTS diaSemana CASCADE;
DROP TABLE IF EXISTS franco CASCADE;
DROP TABLE IF EXISTS minorista CASCADE;
DROP TABLE IF EXISTS mayorista CASCADE;
DROP TABLE IF EXISTS vendedor CASCADE;
DROP TABLE IF EXISTS cliente CASCADE;
DROP TABLE IF EXISTS persona CASCADE;


CREATE TABLE persona(
    DNI INT PRIMARY KEY NOT NULL,
    apellido VARCHAR(20),
    nombre VARCHAR(20),
    direccion VARCHAR(20)
);


CREATE TABLE cliente(
    DNI INT PRIMARY KEY NOT NULL,
    CONSTRAINT dni_fk FOREIGN KEY (DNI)
        REFERENCES persona(DNI)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


CREATE TABLE vendedor(
    DNI INT PRIMARY KEY NOT NULL,
    nro_vendedor INT GENERATED BY DEFAULT AS IDENTITY UNIQUE,
    CONSTRAINT dni_fk FOREIGN KEY (DNI)
        REFERENCES persona(DNI)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


CREATE TABLE mayorista(
    DNI INT PRIMARY KEY NOT NULL,
    descuento NUMERIC(5,2),
    CONSTRAINT dni_fk FOREIGN KEY (DNI)
        REFERENCES cliente(DNI)
        ON DELETE CASCADE
        ON UPDATE CASCADE,
    CONSTRAINT descuento_valido CHECK(descuento >= 0 AND descuento <= 100)
);


CREATE TABLE minorista(
    DNI INT PRIMARY KEY NOT NULL,
    fecha_nacimiento DATE,
    CONSTRAINT dni_fk FOREIGN KEY (DNI)
        REFERENCES cliente(DNI)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


CREATE TABLE franco(
    nro_franco INT PRIMARY KEY,
    fecha DATE,
    observaciones VARCHAR(30),
    DNI_V INT NOT NULL,
    CONSTRAINT dni_v_fk FOREIGN KEY (DNI_V)
        REFERENCES vendedor(DNI)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


CREATE TABLE diaSemana(
    dia VARCHAR(20) PRIMARY KEY NOT NULL
);


CREATE TABLE horario(
    DNI_V INT NOT NULL,
    dia VARCHAR(20) NOT NULL,
    horario_salida TIME,
    horario_entrada TIME,
    CONSTRAINT horario_pk PRIMARY KEY (DNI_V, dia),
    CONSTRAINT dni_v_fk FOREIGN KEY (DNI_V)
        REFERENCES vendedor(DNI)
        ON DELETE CASCADE
        ON UPDATE CASCADE,
    CONSTRAINT dia_fk FOREIGN KEY (dia)
        REFERENCES diaSemana(dia)
        ON DELETE RESTRICT
        ON UPDATE CASCADE
);


CREATE TABLE venta(
    nro_factura INT PRIMARY KEY NOT NULL,
    fecha DATE,
    monto INT,
    DNI_C INT NOT NULL,
    DNI_V INT NOT NULL,
    nro_vendedor INT NOT NULL,
    porcentajeDescuento INT,
    CONSTRAINT dni_c_fk FOREIGN KEY (DNI_C)
        REFERENCES cliente(DNI)
        ON DELETE RESTRICT
        ON UPDATE CASCADE,
    CONSTRAINT dni_v_fk FOREIGN KEY (DNI_V)
        REFERENCES vendedor(DNI)
        ON DELETE RESTRICT
        ON UPDATE CASCADE,
    CONSTRAINT nro_v_fk FOREIGN KEY (nro_vendedor)
        REFERENCES vendedor(nro_vendedor)
        ON DELETE RESTRICT
        ON UPDATE CASCADE,
    CONSTRAINT porcentajeDescuento_valido CHECK(porcentajeDescuento >= 0 AND porcentajeDescuento <= 100)
);


CREATE TABLE proveedor(
    cuit BIGINT PRIMARY KEY NOT NULL,
    telefonoFijo INT,
    direccion VARCHAR(30),
    razonSocial VARCHAR(30)
);


CREATE TABLE mcelular(
    cuit BIGINT NOT NULL,
    celulares BIGINT NOT NULL,
    CONSTRAINT pk_mult PRIMARY KEY (cuit, celulares),
    CONSTRAINT fk_cuit FOREIGN KEY (cuit)
        REFERENCES proveedor(cuit)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


CREATE TABLE zapato(
    modelo VARCHAR(30) NOT NULL,
    color VARCHAR(20) NOT NULL,
    precio INT,
    CONSTRAINT zapato_pk PRIMARY KEY (modelo, color),
    CONSTRAINT precio_correcto CHECK(precio > 0 AND precio <= 200000)
);


CREATE TABLE contiene(
    modelo VARCHAR(30) NOT NULL,
    color VARCHAR(20) NOT NULL,
    cuit BIGINT NOT NULL,
    CONSTRAINT pk_contiene PRIMARY KEY (modelo, color, cuit),
    CONSTRAINT zapato_fk FOREIGN KEY (modelo, color)
        REFERENCES zapato(modelo, color)
        ON DELETE CASCADE
        ON UPDATE CASCADE,
    CONSTRAINT proveedor_fk FOREIGN KEY (cuit)
        REFERENCES proveedor(cuit)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


CREATE TABLE talle(
    nro INT PRIMARY KEY NOT NULL
);


CREATE TABLE tiene(
    modelo VARCHAR(30) NOT NULL,
    color VARCHAR(20) NOT NULL,
    nro INT NOT NULL,
    CONSTRAINT pk_tiene PRIMARY KEY (modelo, color, nro),
    CONSTRAINT zapato_fk FOREIGN KEY (modelo, color)
        REFERENCES zapato(modelo, color)
        ON DELETE RESTRICT
        ON UPDATE CASCADE,
    CONSTRAINT nro_fk FOREIGN KEY (nro)
        REFERENCES talle(nro)
        ON DELETE RESTRICT
        ON UPDATE CASCADE
);


CREATE TABLE posee(
    nro INT NOT NULL,
    modelo VARCHAR(30) NOT NULL,
    color VARCHAR(20) NOT NULL,
    nro_factura INT NOT NULL,
    cantidad INT,
    precio_venta INT,
    CONSTRAINT pk_posee PRIMARY KEY (nro, modelo, color, nro_factura),
    CONSTRAINT tiene_fk FOREIGN KEY (modelo, color, nro)
        REFERENCES tiene(modelo, color, nro)
        ON DELETE RESTRICT
        ON UPDATE CASCADE,
    CONSTRAINT nro_f_fk FOREIGN KEY (nro_factura)
        REFERENCES venta(nro_factura)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


CREATE TABLE auditoria(
    id_auditoria INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    modelo VARCHAR(30) NOT NULL,
    color VARCHAR(20) NOT NULL,
    precio_anterior INT,
    precio_nuevo INT,
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    usuario VARCHAR(50) DEFAULT CURRENT_USER
);


CREATE OR REPLACE FUNCTION auditoria_zapato()
RETURNS TRIGGER AS $$
BEGIN
    IF OLD.precio IS DISTINCT FROM NEW.precio THEN
        INSERT INTO auditoria (modelo, color, precio_anterior, precio_nuevo, fecha, usuario)
        VALUES(OLD.modelo, OLD.color, OLD.precio, NEW.precio, CURRENT_TIMESTAMP, CURRENT_USER);
    END IF;
    RETURN NEW;
END;
$$LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS auditoria_trigger ON zapato;

CREATE TRIGGER auditoria_trigger
AFTER UPDATE OF precio ON zapato
FOR EACH ROW
EXECUTE FUNCTION auditoria_zapato();
